cmake_minimum_required(VERSION 3.20)
project(RaggedMLIR VERSION 0.1.0 LANGUAGES C CXX)

find_package(MLIR REQUIRED CONFIG)
option(RAGGED_ALLOW_UNSUPPORTED_LLVM "Permit a development build with another LLVM version" OFF)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "23.1.3" AND NOT RAGGED_ALLOW_UNSUPPORTED_LLVM)
  message(FATAL_ERROR "Ragged requires LLVM/MLIR 23.1.3; found ${LLVM_PACKAGE_VERSION}. For compatibility experiments only, set RAGGED_ALLOW_UNSUPPORTED_LLVM=ON.")
endif()
include(${PROJECT_SOURCE_DIR}/cmake/LLVMProvenance.cmake)
configure_file(include/Ragged/ToolchainConfig.h.in
               include/Ragged/ToolchainConfig.h @ONLY)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include(HandleLLVMOptions)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LLVM_RUNTIME_OUTPUT_INTDIR ${CMAKE_BINARY_DIR}/bin)
set(LLVM_LIBRARY_OUTPUT_INTDIR ${CMAKE_BINARY_DIR}/lib)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${PROJECT_SOURCE_DIR}/include ${PROJECT_BINARY_DIR}/include)
separate_arguments(RAGGED_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
add_definitions(${RAGGED_LLVM_DEFINITIONS})
add_subdirectory(include/Ragged)
add_subdirectory(lib)
add_subdirectory(tools)
option(RAGGED_INCLUDE_TESTS "Build lit tests" ON)
if(RAGGED_INCLUDE_TESTS)
  enable_testing()
  add_subdirectory(test)
endif()
